#include <iostream>
#include <iomanip>
#include <string>
using namespace std;

int main()
{
    int n;
    cin >> n;
    cout << fixed << setprecision(2);
    for(int i = 0; i < n; i++)
    {
        string name;
        double chinese, math, eng;
        cin >> name >> chinese >> math >> eng;
        double avg = (chinese + math + eng) / 3.0;
        cout << left << setw(10) << name
             << setw(8) << chinese
             << setw(8) << math
             << setw(8) << eng
             << setw(8) << avg;
        if(avg >= 90)
            cout << " Excellent";
        cout << endl;
    }
    return 0;
}


task2 const指针统计元音


#include <iostream>
#include <cstring>
using namespace std;

int countVowels(const char* p)
{
    int cnt = 0;
    while(*p != '\0')
    {
        if(*p == 'a' || *p == 'e' || *p == 'i' || *p == 'o' || *p == 'u')
        {
            cnt++;
        }
        p++;
    }
    return cnt;
}

int main()
{
    char str[100];
    cin.ignore();
    cin.getline(str,100);
    int res = countVowels(str);
    cout << "Number of vowels: " << res << endl;
    return 0;
}


task3 多功能计算器

#include <iostream>
#include <iomanip>
using namespace std;

inline double square(double x)
{
    return x * x;
}

double power(double base, int exponent = 2)
{
    double res = 1.0;
    for(int i = 0; i < exponent; i++)
    {
        res *= base;
    }
    return res;
}

double calc(double a, double b, char op)
{
    if(op == '+') return a + b;
    else if(op == '-') return a - b;
    else if(op == '*') return a * b;
    else if(op == '/')
    {
        if(b == 0)
        {
            cout << "Error: divide by zero" << endl;
            return 0;
        }
        return a / b;
    }
    return 0;
}

double calc(int a, int b, char op)
{
    return calc((double)a, (double)b, op);
}

int main()
{
    cout << fixed << setprecision(2);
    double a,b;
    char op;
    cin >> a >> b >> op;

    cout << "square(" << a << ") = " << square(a) << endl;
    cout << "power(" << a << ") = " << power(a) << endl;
    double ans = calc(a,b,op);
    if(!(op == '/' && b == 0))
    {
        cout << a << " " << op << " " << b << " = " << ans << endl;
    }
    return 0;
}


task4 引用swap交换

#include <iostream>
using namespace std;

void swap(int &x, int &y)
{
    int t = x;
    x = y;
    y = t;
}

int main()
{
    int x,y;
    cin >> x >> y;
    cout << "Before swap: x=" << x << ", y=" << y << endl;
    swap(x,y);
    cout << "After swap: x=" << x << ", y=" << endl;
    return 0;
}
